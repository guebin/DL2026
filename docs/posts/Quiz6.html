b0=0

def l(b0,b1):
    return torch.sum((y-(b0+b1*x))**2)
for dummy in range(1000):
    b0_grad = (l(b0+0.0001,b1) - l(b0,b1))/0.0001

    b1 = b1 - alpha * b1_grad
